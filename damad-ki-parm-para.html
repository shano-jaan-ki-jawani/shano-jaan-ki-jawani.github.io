پرمپرا

ہیلو دوستو، کیسے ہیں آپ؟ امید کرتی ہوں آپ ٹھیک ہوں گے۔ آج میں اپنی زندگی کی ایک ایسی سچی کہانی سنانے جا رہی ہوں جو میری تقدیر بدلنے کا سبب بنی۔ جب میری شادی ہوئی، میرے شوہر ایک انتہائی سیدھے سادھے اور شریف انسان تھے۔ وہ مجھ سے بے پناہ محبت کرتے تھے اور ان کی یہی محبت میری زندگی کا کل اثاثہ تھی۔ شادی کے چھ مہینے بعد میرے ہاں خوشخبری آئی اور اللہ نے ہمیں ایک پیاری سی بیٹی سے نوازا، جس کا نام میرے شوہر نے تبسم رکھا۔ لیکن افسوس، جب تبسم صرف تین مہینے کی ہوئی، تو میرے شوہر اچانک اس دنیا سے رخصت ہو گئے۔ وہ دن میرے لیے کسی قیامت سے کم نہیں تھا۔ اس بڑے اور سنسان گھر میں، میں اپنی تین مہینے کی معصوم بچی کے ساتھ بالکل اکیلی رہ گئی۔ میں نے اپنی بیٹی کو پالنے کے لیے دن رات ایک کر دیے، مگر جب رات کا سناٹا چھاتا، تو مجھے اپنے شوہر کی قربت اور محبت کی بہت یاد ستاتی۔ اسی طرح وقت گزرتا گیا اور بیٹی ایک سال کی ہو گئی۔ وقت کے ساتھ ساتھ میرے اندر ایک جوان عورت کے جذبات انگڑائیاں لینے لگے، لیکن بیٹی کی خاطر میں نے اپنے دل پر پتھر رکھ لیا اور کسی غیر مرد کی طرف آنکھ اٹھا کر بھی نہ دیکھا۔ جب بیٹی تیرہ سال کی ہوئی، تو تنہائی کے بوجھ اور جوان ہوتے جذبات نے مجھے اندر سے بے چین کرنا شروع کر دیا۔ جب بھی میں کسی مرد کو دیکھتی، تو میرے دل میں ایک ساتھی کی شدید تمنا جاگ اٹھتی۔ یہ بے چینی جب حد سے بڑھتی، تو میں خود ہی اپنے جذبات کو کسی طرح پرسکون کرتی اور کچھ دن کے لیے خاموش ہو جاتی۔ جب تبسم اٹھارہ سال کی ہوئی، تو ایک دن اس نے شرماتے ہوئے مجھ سے کہا، "ممی، میں ایک لڑکے سے پیار کرتی ہوں اور اس سے شادی کرنا چاہتی ہوں، میں چاہتی ہوں کہ آپ اس سے ملیں۔ اگر آپ کی اجازت ہو تو میں اسے آج رات کھانے پر بلا لوں؟" میں نے ہنس کر کہا، "ٹھیک ہے بیٹا، ضرور بلا لو۔" تبسم خوش ہو کر اسے لینے چلی گئی۔ رات کو تبسم اس لڑکے کے ساتھ گھر آئی اور بولی، "ممی، یہ وقاص ہے۔" میں نے وقاص کو دیکھا تو وہ ایک انتہائی وجیہہ، صحت مند اور خوبصورت نوجوان تھا۔ ہم سب کھانے کی میز پر بیٹھے۔ دورانِ گفتگو میں نے مسکرا کر پوچھا، "وقاص بیٹا، شادی کا کیا ارادہ ہے؟" وقاص نے احترام سے جواب دیا، "ممی جی، جب آپ حکم کریں۔" میں نے ہنس کر کہا، "پھر تو جلدی ہی کر لو۔" اس پر وقاص نے تبسم کی طرف دیکھا اور کہا، "تم نے ممی کو وہ بات ابھی تک نہیں بتائی؟" میں چونک گئی اور پوچھا، "کیا بات ہے تبسم؟ بتاؤ۔" تبسم شرمانے لگی اور بولی, "ممی، وقاص ہی آپ کو بتائیں گے۔" وقاص نے بات سنبھالتے ہوئے کہا، "ممی جی، اصل میں ہمارے خاندان کی ایک عجیب پرمپرا (رواہت) ہے کہ شادی سے پہلے دولہا کو اپنی ساس کے ساتھ ایک رات گزارنی ہوتی ہے۔ تبسم کو اس پر کوئی اعتراض نہیں ہے، اب آپ بتائیں۔" وقاص کی یہ بات سن کر تو جیسے میرے پیروں تلے سے زمین نکل گئی۔ اتنے میں تبسم کہنے لگی، "ممی پلیز مان جائیں، وقاص بہت اچھے ہیں اور میں انہیں کھونا نہیں چاہتی۔" بیٹی کی یہ بات سن کر میرے اندر برسوں سے سوئے ہوئے جذبات کا ایک طوفان ابل پڑا۔ میری رگوں میں خون کی گردش تیز ہو گئی اور میرا پورا وجود اس نوجوان کی قربت پانے کے لیے بے تاب ہونے لگا، مگر میں نے اپنی خاموشی برقرار رکھی کہ کیا جواب دوں۔ وقاص نے دوبارہ پوچھا، "ممی جی، بتائیے؟" اور تبسم بھی ضد کرنے لگی۔ آخر کار میں نے اپنے جذبات کے آگے ہتھیار ڈال دیے اور کہا، "اچھا ٹھیک ہے، اگر تمہاری یہی پرمپرا ہے تو بتاؤ کب کرنا ہے؟" تبسم خوش ہو کر بولی، "ممی، آج رات ہی آپ یہ رسم پوری کر دیں، کل ہماری شادی ہو جائے گی۔" اتنے میں وقاص نے ہلکے سے اشارے میں پوچھا، "ممی جی، کیا آپ اس خاص رات کے لیے پوری طرح تیار ہیں؟" یہ سنتے ہی میرے اندر کی آگ مزید بھڑک اٹھی۔ تبسم نے وقاص کا ہاتھ پکڑا اور اپنے کمرے کی طرف لے جاتے ہوئے بولی، "ممی، آپ فریش ہو جائیں، تب تک ہم ذرا وقت گزارتے ہیں۔" میرا پورا جسم فرطِ جذبات سے تپ رہا تھا۔ میں اپنے کمرے کے واش روم میں آئی اور آئینے میں خود کو دیکھا۔ میرے شباب کی رعنائیاں اور ابھار اس وقت عروج پر تھے۔ میں نے سوچا کہ جب بیٹی خود راضی ہے تو میں کیوں انکار کروں؟ میں نے جلدی سے خود کو صاف ستھرا اور خوشبودار کیا اور جب تبسم کے کمرے کے پاس سے گزری تو اندر سے ان کی ہنسی اور قربت کی آوازیں آ رہی تھیں۔ میں ڈرائنگ روم میں آ کر ٹی وی دیکھنے لگی۔ کچھ دیر بعد دونوں باہر آئے۔ تبسم نے مسکرا کر وقاص سے کہا، "اب آپ ممی کے روم میں جائیں۔" وقاص نے سحر انگیز نظروں سے مجھے دیکھا اور کہا، "ممی جی، چلیں۔" میں شرماتی ہوئی آگے بڑھی اور وقاص میرے پیچھے آنے لگا۔ کمرے میں داخل ہوتے ہی وقاص نے مجھے پیچھے سے اپنی مضبوط باہوں میں بھر لیا۔ اس کا کسا ہوا جوان وجود جب میرے کولہوں سے ٹکرایا، تو مجھے ایک سحر انگیز احساس ملا۔ دل چاہا کہ وقاص ابھی اسی وقت مجھے اپنے وجود کا حصہ بنا لے۔ وقاص نے اپنے ہاتھ میرے بھرپور ابھاروں پر رکھے اور انہیں دباتے ہوئے کہا، "ممی جی، آپ کا جسم تو بہت گرم ہے۔" میں گھوم کر اس کے سامنے آئی، اسے اپنی باہوں میں جکڑ لیا اور پاگلوں کی طرح چومنے لگی۔ وقاص نے ایک آہ بھری اور اپنے ہونٹ میرے ہونٹوں پر رکھ دیے۔ ہم دونوں ایک دوسرے کے سانسوں کی گرمی اور زبان کا رس چوسنے لگے۔ اسی مستی میں، میں نے اس کے پاجامے کے اندر ہاتھ ڈالا تو وہاں ایک انتہائی مضبوط، طویل اور جاندار مردانگی میرا انتظار کر رہی تھی۔ میں نے اس کا ناڑا کھولا تو اس کا لباس نیچے گر گیا اور اس نے بھی میری قمیض اتار دی۔ وہ میرے ابھاروں کو منہ میں لے کر چوسنے اور مسلنے لگا۔ میں نے بھی جلدی سے اس کی قمیض اتاری اور اس کے لوہے جیسے سخت وجود کو اپنے ہاتھوں میں لے کر محسوس کیا، وہ واقعی ایک طاقتور مرد تھا۔ وقاص نے مجھے نرم بستر پر لیٹا دیا اور میرا زیریں لباس بھی اتار دیا۔ میری بے داغ اور سڈول جوانی کو دیکھ کر اس کی آنکھیں چمک اٹھیں، وہ بولا، "ممی جی، آپ تو کمال ہیں۔" وہ نیچے جھکا اور میرے وجود کے سب سے نازک اور حساس حصے کو چومنے اور چاٹنے لگا۔ اس کی گرم زبان جب میرے نازک حصے کے اندر تک گئی، تو لذت کی ایک لہر میرے پورے جسم میں دوڑ گئی۔ میں مستی میں اپنے پیروں سے اس کے وجود کو سہلا رہی تھی۔ پھر وقاص نے مجھے اپنی جوان مردانگی دکھائی اور پوچھا، "ممی جی، پسند تو آئی نہ؟" میں نے دیوانہ وار کہا، "بہت زیادہ۔" اور اسے بستر پر لٹا کر خود اس کے اوپر آ گئی اور اس کی مردانگی کو اپنے منہ میں لے کر چوسنے لگی۔ وقاص مستی میں میری تعریفیں کر رہا تھا۔ کچھ دیر بعد اس نے مجھے سیدھا لٹایا اور میری ٹانگوں کو اٹھا کر میرے کندھوں تک کر دیا، جس سے میرا پورا نچلا وجود اوپر اٹھ گیا۔ وقاص نے ایک زوردار جھٹکے سے اپنی پوری مردانگی میرے نازک حصے کے اندر اتار دی۔ کمرے میں میری سسکیاں اور ہمارے جسموں کے ٹکرانے کی پٹاخ پٹاخ کی آوازیں گونجنے لگیں۔ میں ایک ایسے لازوال مزے میں تھی جس کے لیے میں برسوں سے ترس رہی تھی۔ پھر اس نے مجھے گھوڑی بنایا اور پیچھے سے ایک ایک انچ اندر اتار کر مجھے لذت کے آسمان پر پہنچا دیا۔ میں بھی اس کے ہر جھٹکے کا بھرپور ساتھ دے رہی تھی۔ پھر میں اس کے اوپر آئی اور خود کو اس کی مردانگی پر گرا کر تڑپنے لگی۔ وقاص چلایا، "ممی جی، آپ کے اندر تو بلا کا رس ہے!" اسی طرح ہمیں دو گھنٹے گزر گئے، یہاں تک کہ میرے وجود نے اپنا سارا رس چھوڑ دیا اور وقاص نے بھی اپنا گرم مادہ میرے اندر انڈیل دیا۔ ہم کچھ دیر سکون سے لیٹے رہے، پھر میں نے نائٹی پہن لی مگر وقاص اسی حالت میں باہر آگیا۔ تبسم باہر صوفے پر بیٹھی تھی، وہ وقاص سے لپٹ گئی اور بولی، "وقاص، اب تو ممی کی رسم پوری ہو گئی، اب شادی پکی؟" وقاص نے کہا, "ہاں بالکل۔" میں کچن میں چائے بنانے لگی تو وہ دونوں صوفے پر ہی ایک دوسرے کے ساتھ مست ہو گئے۔ انہیں دیکھ کر میرا وجود پھر سے پگھلنے لگا۔ چائے پی کر تبسم وقاص کو اپنے کمرے میں لے گئی، مگر میرا برا حال تھا کہ وقاص کب واپس آئے۔ کچھ دیر بعد وقاص میرے کمرے میں لوٹ آیا اور پھر صبح تک ہم دونوں محبت کی وادیوں میں کھوئے رہے۔ دوسرے دن وقاص اور تبسم کی شادی ہو گئی، لیکن ہماری کہانی یہاں ختم نہیں ہوئی۔ وقاص نے تبسم سے کہا کہ ہم تینوں مل کر سہاگ رات منائیں گے اور تبسم مان گئی۔ اب ایک ہی کمرے میں، میں، میری بیٹی اور وقاص رات بھر مستیاں کرتے۔ میرا تو دل کرتا کہ وقاص کبھی میرے وجود سے الگ ہی نہ ہو۔ ایک رات اس نے مجھے الٹا لٹا کر میرے پچھلے حصے (گانڈ) میں جانے کی خواہش کی، میں تو اس کی محبت میں اس حد تک پاگل تھی کہ میں نے کہا، "چاہے جتنا درد ہو، تم مجھے پورا حاصل کر لو۔"کچھ ہی دنوں میں تبسم امید سے ہو گئی، وہ پچھلے حصے میں جانے سے ڈرتی تھی مگر میں وقاص کو زندگی کے سارے مزے دیتی تھی۔ پھر تبسم کے ہاں بیٹا ہوا، اور کچھ مہینوں بعد وقاص کی محبت کی نشانی میرے پیٹ میں بھی ٹھہر گئی۔ میں نے اسے گرانے کا فیصلہ کیا، مگر تبسم اور وقاص نے میرا ہاتھ پکڑ لیا اور منع کر دیا۔۔۔اگلے پارٹ کیلئے یہاں کلک کریں 👇
